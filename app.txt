from flask import Flask, request, jsonify
from flask_cors import CORS
from transformers import AutoTokenizer, AutoModel
import torch
import torch.nn as nn
import torch.nn.functional as F
from torch.optim import AdamW
import pandas as pd
from sklearn.preprocessing import LabelEncoder
from torch.utils.data import DataLoader, Dataset
from sklearn.utils import shuffle
import numpy as np
from sklearn.linear_model import LogisticRegression

# Setup Flask app
app = Flask(__name__)
CORS(app)

MODEL_NAME = "google-bert/bert-base-uncased"
tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME)
device = torch.device("cuda" if torch.cuda.is_available() else "cpu")

# Globals
model = None
label_encoder = LabelEncoder()
is_training = False

# Dummy Logistic Regression (for structured survey data)
logistic_regression = LogisticRegression()
logistic_regression.classes_ = np.array([0, 1, 2])  # Low, Moderate, High
logistic_regression.coef_ = np.array([[0.5] * 9, [1.0] * 9, [1.5] * 9])  # Replace with real weights
logistic_regression.intercept_ = np.array([0, 0, 0])

def normalize(scores):
    # Normalize PHQ-9/GAD-7 scores to [0,1]
    return np.array(scores) / (len(scores) * 3)

# Custom Dataset
class EmotionDataset(Dataset):
    def __init__(self, texts, labels):
        self.texts = texts
        self.labels = labels

    def __len__(self):
        return len(self.texts)

    def __getitem__(self, idx):
        encoding = tokenizer(
            self.texts[idx],
            return_tensors='pt',
            truncation=True,
            padding='max_length',
            max_length=128
        )
        input_ids = encoding['input_ids'].squeeze(0)
        attention_mask = encoding['attention_mask'].squeeze(0)
        return {
            'input_ids': input_ids,
            'attention_mask': attention_mask,
            'label': torch.tensor(self.labels[idx], dtype=torch.long)
        }

# Model
class EmotionClassifier(nn.Module):
    def __init__(self, num_labels):
        super(EmotionClassifier, self).__init__()
        self.bert = AutoModel.from_pretrained(MODEL_NAME)
        self.dropout = nn.Dropout(0.3)
        self.classifier = nn.Linear(self.bert.config.hidden_size, num_labels)

    def forward(self, input_ids, attention_mask, labels=None):
        outputs = self.bert(input_ids=input_ids, attention_mask=attention_mask)
        cls_output = outputs.last_hidden_state[:, 0]
        logits = self.classifier(self.dropout(cls_output))
        loss = F.cross_entropy(logits, labels) if labels is not None else None
        return {"logits": logits, "loss": loss}

# --- Training Endpoint ---
@app.route('/train', methods=['POST'])
def train():
    global model, label_encoder, is_training

    if is_training:
        return jsonify({"status": "Training already in progress"}), 429

    try:
        is_training = True

        if 'file' not in request.files:
            return jsonify({"error": "CSV file missing"}), 400

        file = request.files['file']
        df = pd.read_csv(file)

        if 'text' not in df.columns or 'label' not in df.columns:
            return jsonify({"error": "CSV must contain 'text' and 'label' columns"}), 400

        df = shuffle(df)
        texts = df['text'].astype(str).tolist()
        labels = label_encoder.fit_transform(df['label'].astype(str))
        num_labels = len(label_encoder.classes_)

        dataset = EmotionDataset(texts, labels)
        dataloader = DataLoader(dataset, batch_size=8, shuffle=True)

        model = EmotionClassifier(num_labels).to(device)
        optimizer = AdamW(model.parameters(), lr=2e-5)
        model.train()

        for epoch in range(3):
            total_loss = 0
            for batch in dataloader:
                input_ids = batch['input_ids'].to(device)
                attention_mask = batch['attention_mask'].to(device)
                label = batch['label'].to(device)

                output = model(input_ids=input_ids, attention_mask=attention_mask, labels=label)
                loss = output['loss']

                optimizer.zero_grad()
                loss.backward()
                optimizer.step()

                total_loss += loss.item()

            print(f"Epoch {epoch + 1} Loss: {total_loss:.4f}")

        return jsonify({"message": "Training complete", "labels": label_encoder.classes_.tolist()})

    except Exception as e:
        return jsonify({"error": str(e)}), 500

    finally:
        is_training = False

# Training status
@app.route('/training_status', methods=['GET'])
def training_status():
    return jsonify({"is_training": is_training})

# Inference
@app.route('/analyze', methods=['POST'])
def analyze():
    global model, label_encoder

    try:
        if model is None:
            return jsonify({"error": "Model not trained yet"}), 400

        data = request.get_json()
        text = data.get('text', '').strip()

        if not text:
            return jsonify({"error": "No text provided"}), 400

        model.eval()
        with torch.no_grad():
            inputs = tokenizer(text, return_tensors='pt', truncation=True, padding='max_length', max_length=128)
            input_ids = inputs['input_ids'].to(device)
            attention_mask = inputs['attention_mask'].to(device)

            output = model(input_ids=input_ids, attention_mask=attention_mask)
            logits = output['logits']
            probs = F.softmax(logits, dim=-1)
            top_prob, top_idx = torch.max(probs, dim=1)

            predicted_label = label_encoder.inverse_transform([top_idx.item()])[0]
            NEGATIVE = {"anger", "sadness", "fear", "disgust", "guilt", "shame"}
            is_negative = predicted_label.lower() in NEGATIVE and top_prob.item() > 0.6

            return jsonify({
                "label": predicted_label,
                "score": round(top_prob.item(), 4),
                "is_negative": is_negative
            })

    except Exception as e:
        return jsonify({"error": str(e)}), 500

# Hybrid Risk Analysis
@app.route('/hybrid_risk', methods=['POST'])
def hybrid_risk():
    try:
        data = request.get_json()
        survey_data = data.get("survey_data")  # List of 9 PHQ-9 answers
        user_post = data.get("user_post", "")

        if not survey_data or len(survey_data) != 9:
            return jsonify({"error": "Invalid survey data"}), 400
        if not user_post.strip():
            return jsonify({"error": "User post is required"}), 400

        # Step 1: LR on structured data
        norm_scores = normalize(survey_data).reshape(1, -1)
        lr_probs = logistic_regression.predict_proba(norm_scores)[0]  # [Low, Moderate, High]

        # Step 2: BERT CLS anomaly score
        tokens = tokenizer(user_post, return_tensors="pt", truncation=True, padding="max_length", max_length=128)
        tokens = {k: v.to(device) for k, v in tokens.items()}

        with torch.no_grad():
            cls_embedding = model.bert(**tokens).last_hidden_state[:, 0, :]
            logits = model.classifier(model.dropout(cls_embedding))
            anomaly_score = torch.sigmoid(logits).cpu().numpy()[0][0]

        # Step 3: Hybrid integration
        hybrid_probs = np.array([
            0.7 * lr_probs[0],                        # Low
            0.7 * lr_probs[1],                        # Moderate
            0.7 * lr_probs[2] + 0.3 * anomaly_score   # High + BERT influence
        ])

        risk_levels = ["Low", "Moderate", "High"]
        risk_index = np.argmax(hybrid_probs)
        final_risk = risk_levels[risk_index]

        return jsonify({
            "risk_level": final_risk,
            "hybrid_probs": hybrid_probs.tolist(),
            "lr_probs": lr_probs.tolist(),
            "bert_score": float(anomaly_score)
        })

    except Exception as e:
        return jsonify({"error": str(e)}), 500

if __name__ == '__main__':
    app.run(debug=True)
